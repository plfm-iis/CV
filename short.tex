\section*{Research}
\begin{itemize}
  \item \textbf{Interests:} Algorithmic formal verification; quantum programs (correctness, data structures, semantics).
  \item \textbf{Selected results:} Founder of AutoQ (automated verification of quantum programs); contributor to the Medusa quantum simulator.
  \item \textbf{Ongoing work:} Verification of FTQC, QECC, QSVT-based quantum programs, and programs with unbounded qubits.
\end{itemize}
